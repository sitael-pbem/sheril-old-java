# quelque chose à détecter. Chaque ligne non vide et non commentée doit
# apparaître telle quelle dans le dump du dernier tour.
#
# "tour" vaut TOURS+1 : dump-tour-N.txt est capturé après le newRound du
# tour N, et le premier newRound (inscription, avant la boucle des tours)
# fait déjà passer le compteur interne de 0 à 1.
tour = 7
commandant.1.nom = Alpha
commandant.2.nom = Beta
commandant.1.humain = true
commandant.2.humain = true
#
# Effet de jeu : la puissance de la flotte 0 de BETA (le défenseur) après le
# combat du tour 5. Pas le nombre de ses vaisseaux : dix par tour1-4, un
# combat réel ne fait tomber ce nombre qu'à 3 (perte massive), une valeur
# trop dépendante du tirage pour rester un bon témoin structurel. La
# puissance, elle, encaisse tout dégât, même partiel : 250 avant combat,
# 75 après, stable ensuite (le combat ne se rejoue pas une fois la
# directive consommée). Cette ligne serait fausse si :
# - le marqueur de mouvement ne se résolvait plus ou pointait ailleurs
#   (piège de colonnes POSX/POSY, voir l'en-tête de test/harness/lib.sh) ;
# - la directive d'attaque ciblée sur Beta (8 + 100 * numéro, voir
#   turn-3.sql) redevenait une DIRECTIVE_FLOTTE_ATTAQUE_TOUTE_FLOTTES : dans
#   ce cas la flotte d'Alpha engage la garnison neutre du système (toujours
#   présente à la capitale d'un commandant) et jamais Beta, dont les 8
#   lignes de flotte restent alors identiques d'un tour à l'autre, y compris
#   cette ligne (vu rouge : puissance restée à 250 dans cette configuration,
#   confirmé en rejouant ce scénario avec DIRECTIVE = 3 avant cette
#   correction).
commandant.2.flotte.0.puissance = 75
